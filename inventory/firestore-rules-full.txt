rules_version = '2';
service cloud.firestore {
  match /databases/{db}/documents {
    function me() { return get(/databases/$(db)/documents/users/$(request.auth.uid)).data; }
    function signed() { return request.auth != null && exists(/databases/$(db)/documents/users/$(request.auth.uid)); }
    function isAdmin() { return signed() && me().role == 'admin'; }

    function invMe() { return get(/databases/$(db)/documents/inv_users/$(request.auth.uid)).data; }
    function invSigned() { return request.auth != null && exists(/databases/$(db)/documents/inv_users/$(request.auth.uid)); }
    function invAdmin() { return (invSigned() && invMe().role == 'admin') || isAdmin(); }

    match /users/{u} {
      allow read: if request.auth != null && (request.auth.uid == u || isAdmin());
      allow write: if isAdmin();
    }
    match /config/{d} {
      allow read: if signed();
      allow write: if isAdmin();
    }
    match /headcount/{d} {
      allow write: if isAdmin();
      allow read: if isAdmin() || (signed() && me().branch == d);
    }
    match /sales/{s} {
      allow read: if signed();
      allow create, update: if isAdmin() ||
        (signed() && me().role == 'user'
        && me().branch == request.resource.data.branch
        && s == request.resource.data.branch + '_' + request.resource.data.date);
      allow delete: if isAdmin();
    }

    match /inv_users/{u} {
      allow read: if request.auth != null && (request.auth.uid == u || invAdmin());
      allow write: if invAdmin();
    }
    match /inv_cfg/{d} {
      allow read: if invSigned() || isAdmin();
      allow write: if invAdmin();
    }
    match /inv_imgs/{d} {
      allow read: if invSigned() || isAdmin();
      allow write: if invAdmin();
    }
    match /inv_exp/{d} {
      allow read, write: if invAdmin();
    }
    match /inv_counts/{d} {
      allow read: if invAdmin() || (invSigned() && invMe().branch == resource.data.b);
      allow create, update: if invAdmin() || (invSigned() && invMe().role == 'user' && invMe().branch == request.resource.data.b);
    }
  }
}
